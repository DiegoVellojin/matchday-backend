# ==============================================================================
# MATCHDAY BACKEND - ARQUITECTURA DE SOFTWARE
# ==============================================================================
# Este proyecto implementa una Arquitectura Limpia (Clean Architecture) orientada
# a Dominios (Domain-Driven Design Lite), permitiendo un código altamente 
# escalable, testeable e independiente de frameworks externos.
# ==============================================================================

backend/
├── app/
│   ├── main.py              # Punto de entrada FastAPI: Inicialización y registro de rutas globales.
│   ├── core/                # Componentes Transversales: Lógica compartida por toda la aplicación.
│   │   ├── config.py        # Gestión de variables de entorno y constantes con Pydantic Settings.
│   │   ├── database.py      # Configuración de Engine y SessionLocal de SQLAlchemy para Neon.db.
│   │   ├── security.py      # Criptografía: Generación/Validación de JWT y lógica de OAuth2 con Google.
│   │   └── mail.py          # Motor de Mensajería: Configuración SMTP para envío de correos HTML (Reset Password).
│   ├── modules/             # Capa de Negocio: Dividida por dominios funcionales (Features).
│   │   ├── auth/            # Seguridad y Sesiones: Registro local, Google Login y recuperación.
│   │   ├── users/           # Perfiles y Estadísticas: Gestión de identidad y rendimiento del jugador.
│   │   ├── matches/         # Partidos: Orquestación de creación, convocatoria y estados de juego.
│   │   ├── teams/           # Equipos: Estructura de grupos, rosters y capitanías.
│   │   ├── payments/        # Finanzas: Control de recaudos, deudas y validación de comprobantes.
│   │   └── rankings/        # Competición: Algoritmos de puntuación y tablas de posiciones.
│   └── shared/              # Utilidades Genéricas: Helpers, formateadores y decoradores comunes.
├── requirements.txt         # Manifiesto de dependencias (FastAPI, Pydantic, SQLAlchemy, Bcrypt, Jose).
├── .env                     # Secrets: Credenciales sensibles (Excluido de Git por seguridad).
└── MatchDay_db.sql          # Snapshot SQL: Estructura actual de la base de datos para referencia.

# ------------------------------------------------------------------------------
# ANATOMÍA DE UN MÓDULO (Ejemplo: auth/)
# ------------------------------------------------------------------------------
# Cada módulo se divide en 4 capas para garantizar la separación de conceptos:
# ------------------------------------------------------------------------------

modules/auth/
├── domain/                  # CAPA 1: EL CORAZÓN (Lógica Pura)
│   ├── entities.py          # Modelos de dominio (Objetos de negocio puros, sin ORM).
│   ├── exceptions.py        # Errores específicos del dominio (Ej: InvalidCredentials).
│   └── interfaces.py        # Contratos (Abstracciones): Definición de qué debe hacer el repositorio.
├── application/             # CAPA 2: CASOS DE USO (Orquestación)
│   └── use_cases.py         # Implementación de la lógica de negocio paso a paso.
├── infrastructure/          # CAPA 3: IMPLEMENTACIÓN (Detalles Técnicos)
│   ├── orm_models.py        # Definición de tablas SQLAlchemy (Esquema físico en BD).
│   └── repositories_impl.py # Código real que habla con la DB usando las interfaces del dominio.
└── presentation/            # CAPA 4: PUERTA DE ENTRADA (Framework)
    ├── router.py            # Controladores FastAPI: Definición de rutas, métodos y códigos HTTP.
    └── schemas.py           # DTOs (Data Transfer Objects): Validaciones Pydantic de entrada y salida.

# ==============================================================================
# REGLA DE ORO: Las dependencias siempre apuntan HACIA ADENTRO.
# El dominio nunca debe conocer nada de FastAPI ni de SQLAlchemy.
# ==============================================================================
